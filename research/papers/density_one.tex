\begin{abstract}
Let $\pi_1(x)$ count integers below $x$ whose $3x{+}1$ orbit reaches $1$, and let
$\gamma(k) = \log_2 \lam^*(k)$ be the density exponent produced by the
Krasikov--Lagarias program family $L_k^{NT}(\lam)$ (Acta Arith.\ 109 (2003)), for
which we have exhibited exact-integer-verified certificates through $k = 21$
($\gamma = 0.9184$, $3.49$ billion constraints). This manuscript assembles a proof
program for $\lim_k \gamma(k) = 1$, i.e.\ $\pi_1(x) \ge x^{1-\varepsilon}$ for
every $\varepsilon > 0$ --- the hope expressed at the end of the original paper.
The program's every \emph{exponent} is proved by elementary means: a
\emph{Freshness Lemma} (each feed step of the certificate's selected-feed forest
reads one fresh ternary digit, bijectively) gives the exact chain-survival law
$(2/3)^g$ and the counting envelope $((B_1{+}B_3)/3)^g \le (3/4)^g$; a
\emph{Saturation Lemma} (the constant $54 = 2\cdot 3^3$ caps every desert cascade
at transmitted depth $2$) bounds all single-structure penalties; a trivial
arithmetic bound caps the positive side of every multiscale increment at
$\log_2 3$; and a \emph{density-beats-tilt} comparison ($\ln 3 / 1.24 > \ln 2$)
controls the flow tilt of the counting measure. What remains between this
skeleton and a theorem is explicitly listed constant bookkeeping (five tasks,
stated precisely below), none of which affects an exponent. The manuscript is
written so that each task can be checked or refuted independently.
\end{abstract}

\section{Setting and main claims}

Fix $k \ge 4$ and $\lam \in (1, 2]$. Classes are indexed by $i \in [0, N)$,
$N = 3^{k-1}$; the class map is $m = 3i+2 \bmod 3^k$. Write $\alpha = \log_2 3$,
$A = \lam^{-2}$, $B_1 = \lam^{\alpha-2}$, $B_3 = \lam^{\alpha-1}$. The
Krasikov--Lagarias operator is
\[
(Fv)(i) \;=\; A\,v(4i{+}2 \bmod N) \;+\;
\begin{cases}
B_1\,\bar v\big(4\lfloor i/3\rfloor \bmod N/3\big) & i \equiv 0 \ (3)\quad(\mathrm{D1})\\
0 & i \equiv 1 \ (3)\quad(\mathrm{D2})\\
B_3\,\bar v\big(2\lfloor i/3\rfloor{+}1 \bmod N/3\big) & i \equiv 2 \ (3)\quad(\mathrm{D3})
\end{cases}
\]
with $\bar v(r) = \min\{v(r), v(r+N/3), v(r+2N/3)\}$. Feasibility of $v \le Fv$,
$v > 0$ at parameter $\lam$ yields $\pi_1(x) > x^{\log_2\lam}$ by
\cite[Thm.~2.2]{KL}. Let $v$ denote the Perron vector at the feasibility edge
$\lam^*(k)$ (growth $\rho = 1$), $F = \log_2 v$ the log-field,
$\pbar = 1 - \lam^{-2}/\rho$ the exact feed share (the backbone $i \mapsto 4i+2$
is a permutation, so its share of total flow is exactly $\lam^{-2}/\rho$).

\begin{theorem}[Main claim, modulo Tasks 1--5]\label{thm:main}
$\lim_{k\to\infty} \gamma(k) = 1$; consequently $\pi_1(x) \ge x^{1-\varepsilon}$
for every $\varepsilon > 0$.
\end{theorem}

The logical chain is: bounded multiscale variance of $F$ uniformly in $k$
(Summation Theorem) $\Rightarrow$ geometric decay of feed-domination chain flow
(Lemma D) $\Rightarrow$ flow-$L^2$ attenuation $\kappa \le \kappa_{\max} < 1$
$\Rightarrow$ min-loss $q(k) \to 1$ $\Rightarrow$ $\gamma(k) \to 1$ via the
edge-rate identity $d\gamma/dq = 1/\ln(4/3)$ \PROVED{} \cite{G1}. Sections
2--6 prove every exponent entering this chain; Section 7 records a shorter
conditional route found during assembly (the \emph{endpoint route}:
geometric decay of a single scalar sequence already implies
$\gamma \to 1$ through the proved transfer legs, bypassing the Summation
Theorem and the chain recursion); Section 8 lists the bookkeeping.

\section{The selected-feed forest and the Freshness Lemma}

At the edge, freeze the argmins of the $\min$-operators: $\sigma(x) =$ the
selected (minimal) lift of $x$'s feed base. The map $\sigma$ generates the
\emph{selected-feed graph} on D1/D3 classes.

\begin{lemma}[Type rigidity]\label{lem:type} \PROVED{}
The branch type of $\sigma(x)$ equals the type of its base: all three lifts of a
base $r$ share $r \bmod 3$, since $N/3 \equiv 0 \pmod 3$ for $k \ge 4$; hence the
type walk of the $\sigma$-cascade is independent of the argmin selection.
\end{lemma}

\begin{lemma}[Freshness]\label{lem:fresh} \PROVED{}
Let $W$ be the base walk $x \mapsto 4\lfloor x/3\rfloor \bmod N/3$ (D1) or
$x \mapsto 2\lfloor x/3\rfloor + 1 \bmod N/3$ (D3). Then
\[
W^j(x) \;\equiv\; u_j \,\mathrm{digit}_j(x) + c_j(\text{prefix}) \pmod 3,
\qquad u_j \in \{1,2\},
\]
i.e.\ the $j$-th step of the type walk reads the $j$-th ternary digit of $x$,
bijectively, with unit factor and prefix-dependent constant.
\end{lemma}
\begin{proof}
Each step divides by $3$ (a digit shift) and applies an affine map with unit
multiplier ($4$ or $2$, both invertible mod every power of $3$); composing $j$
steps yields an affine function of $\lfloor x/3^j \rfloor$ with unit linear
coefficient, whose reduction mod $3$ is a unit multiple of $\mathrm{digit}_j(x)$
plus a constant determined by the earlier digits (which fixed the branch choices).
Machine check: bijectivity and prefix-locality verified exhaustively for all
prefixes of length $\le 4$ at $k = 10$.
\end{proof}

\begin{corollary}[Absorption law]\label{cor:absorb} \PROVED{}
The counting density of classes supporting a $\sigma$-chain of length $\ge g$ is
exactly $(2/3)^g$, uniformly in $k$; surviving branch types are i.i.d.\ uniform
on $\{D1, D3\}$; and the selected-feed graph is a forest into D2 absorbers
(zero cycles; verified exhaustively at $k=11$; moreover any cycle would satisfy
$\prod \varphi = \prod b = \lam^{\alpha A - B}$ by telescoping, so B3-rich cycles
are excluded by $\varphi < 1$ and all cycles by the irrationality of $\alpha$
combined with $\varphi<1$ strictly).
\end{corollary}

\begin{corollary}[Counting envelope]\label{cor:envelope} \PROVED{}
$\displaystyle \frac1N \sum_{\text{$g$-chains}} \prod b \;=\;
\Big(\frac{B_1+B_3}{3}\Big)^{g}$, and
$\frac{B_1+B_3}{3} = \frac{2^{\alpha-2}+2^{\alpha-1}}{3} = \frac34$ exactly at
$\lam = 2$; on the whole range $\lam \le 2$, $(B_1+B_3)/(3\rho) \le 0.79$.
\end{corollary}

\section{The Saturation Lemma}

\begin{lemma}[Saturation]\label{lem:sat} \PROVED{}
For $m \equiv -4 \pmod{3^j}$, $j \ge 2$: $m$ is D2 and $4m$ is D1, and the
inherited desert depth of the feed target is
\[
v_3\big(r_1(4m) + 4\big) \;=\; \min(j, 3) - 1 \qquad (j \ne 3),
\]
since $r_1(4m)+4 = \big(16(m+4) - 54\big)/3$ and $v_3(54) = 3$. At the critical
stratum $j = 3$, writing $m+4 = 27t$ with $3 \nmid t$, the transmitted depth is
exactly $2 + v_3(16t-2)$, and $v_3(16t-2) = s$ has density exactly $3^{-s}$.
Consequently every desert touches at most finitely many feed generations at
transmitted depth $\le 2$ and its total value penalty is bounded by an absolute
constant. (Machine checks: $40{,}000/40{,}000$ at the $j=3$ stratum; all sampled
$j \in \{2,4,\dots,8\}$.)
\end{lemma}

\section{Multiscale decomposition and the counting-variance bound}

Let $M_p(i) = \E[F \mid i \bmod 3^p]$ (counting-measure block means),
$X_p = M_p - M_{p-1}$, so $F = M_0 + \sum_p X_p$ with $\E[X_p] = 0$.

\begin{lemma}[Block-equation self-similarity]\label{lem:block} \PROVED{}
Let $V_p(c) = \E[v \mid i \equiv c \bmod 3^p]$ and
$\bar V_{p-1}(r) = \E[\bar v \mid \cdot \equiv r \bmod 3^{p-1}]$. Then for
every $2 \le p \le k-1$, exactly,
\[
V_p(c) \;=\; A\,V_p\big(4c{+}2 \bmod 3^p\big) \;+\;
b(c)\,\bar V_{p-1}\big(R(c) \bmod 3^{p-1}\big),
\]
i.e.\ \emph{the block-mean field satisfies the Krasikov--Lagarias equation
with the feed input one aggregation level down.}
\end{lemma}
\begin{proof}
Take conditional expectations of the fixed-point equation over the block
$\{i \equiv c\}$. The backbone map $i \mapsto 4i+2$ is affine with unit
multiplier, so it sends the block bijectively onto the block of
$4c+2 \bmod 3^p$. The branch type $i \bmod 3$ and hence $b$ is constant on
the block (and $c \bmod 9$ determines D1/D2/D3 for $p \ge 2$). The feed map
$R$ is affine and divides the index by $3$ once, so it sends the mod-$3^p$
block onto a \emph{single} mod-$3^{p-1}$ block, hitting each member once
(unit multiplier again). Linearity of $\E$ finishes.
(Machine check at $k=13$, now for \emph{all} $2 \le p \le 12$: maximal
relative error $2.2\times 10^{-12}$ at every level; script
194\_tower\_task2.py.)
\end{proof}

\begin{remark}[The tower]\label{rem:tower}
Lemma~\ref{lem:block} is the precise form of the generation-versus-level
identification: the multiscale structure of the depth-$k$ field is a
\emph{tower of lower-depth K--L systems} (exact at the level of equations;
measured $\mathrm{corr}(\log V_p, \log v^{(p+1)}) = 0.989/0.995$ at
$p = 5/7$ against the independent depth-$(p{+}1)$ Perron vectors, the
residual deviation reflecting the $\lam$ mismatch between edges). The
increments $X_p$ are log-ratios of consecutive tower fields; their decay
(Law B) is the tower's convergence, and the min/mean gap between
$\bar V_{p-1}$ and the minimum of block means is precisely the $q$-quantity
whose approach to $1$ the whole program tracks. Together with the
constant-lift embedding (companion log, Lemma S1: $\lam^*$ monotone in
depth), this grounds the hierarchical reading used throughout.
\end{remark}

\begin{definition}[Tower decomposition]\label{def:tower}
Set $\tilde X_p = \log_2 V_p - \log_2 V_{p-1}$ (log-ratios of consecutive
block-mean fields). Then $F = \log_2 V_0 + \sum_{p} \tilde X_p$ \emph{exactly}
(telescoping; $V_{k-1} = v$), and each level obeys its own exact equation
(Lemma~\ref{lem:block}).
\end{definition}

\begin{lemma}[Bounded tower increments]\label{lem:pos}
(i) \PROVED{} $\tilde X_p \le \log_2 3$ pointwise: $V_{p-1}$ is the mean of
three positive sub-block means, hence $\ge V_p/3$.
(ii) $\tilde X_p \ge -C_-$ with $C_-$ absolute. \emph{Proof outline
(two proved ingredients + one damping recursion):}
$\tilde X_p(c) = \log_2 \frac{3 V_p(c)}{V_p(c_0)+V_p(c_1)+V_p(c_2)}$ over
the sibling triple at digit $p{-}1$, so it suffices to bound the sibling
spread of the block field. First, \emph{no injection at the top digit}:
siblings share their branch type and hence their coefficients exactly
(adding $t\cdot 3^{p-1}$ does not change $c \bmod 9$ for $p \ge 3$), so by
the Block-Equation Lemma their equations differ only through downstream
values --- backbone images (again a sibling triple) and feed images (a
sibling triple one level down, by digit consumption). Second, the
resulting spread recursion has total weight $< 1$
($\lam^{-2}/\rho + \pbar = 1$ split over strictly deeper spreads with
factor $\le \pbar$ per feed descent), so sibling spread is a damped sum of
injection at lower digits and \emph{saturates}: spread
$\le C_{\mathrm{inj}}^{1/2}\,\pbar/(1-\pbar)$-type bounds, uniformly in
$k$. (Measured: the top-scale lift spread is $0.88$ bits, flat in $k$ from
$11$ to $15$; block-level deserts last one step by the period-3 cycle and
cost at most $\log_2(\rho\lam^2) = 1.73$, with the measured worst
increment $1.48$.) Hence $C_- \le \log_2 3 + \text{saturated spread}$.
\emph{Measured:} $\max_p \sup |\tilde X_p| = 1.48$ at $k=13$, decreasing
in $p$ (from $1.48$ at $p{=}1$ to $0.63$ at $p{=}11$) --- both sides
bounded, in sharp contrast to the martingale increments $X_p$ whose
negative side grows with $k$.
\end{lemma}

\begin{remark}[Why the tower route wins]\label{rem:towerwins}
Three measured facts (k=13) make the tower decomposition strictly more
convenient than the martingale one: (a) $\Var(\tilde X_p)$ decays with
mid-profile ratio $0.66$--$0.71 \approx \pbar$ (Law B holds verbatim);
(b) increments are pointwise bounded (Lemma~\ref{lem:pos}), which makes
tilt stability elementary --- no tail analysis; (c) the cross-correlations
are \emph{universally negative}: all $55$ of $55$ pairs
$(\tilde X_p, \tilde X_q)$, $p < q$, have negative correlation, decaying
smoothly in the lag ($-0.06$ at lag $1$ to $-0.006$ at lag $5$), so that
$\sum_{p,q} \mathrm{Cov} = 1.3796 \approx \Var(F) = 1.3821$ while
$\sum_p \Var(\tilde X_p) = 1.573$: the summation upper bound
$\Var(F) \le \sum_p \Var(\tilde X_p)$ holds with every cross-term helping.
Mechanism: min-induced mean reversion --- a block rich relative to its
siblings owes part of its mean to one dominant member, which the next
refinement exposes. The bookkeeping task is thus the ONE-SIDED bound
$\mathrm{Cov}(\tilde X_p, \tilde X_q) \le c_0\,\mathrm{env}^{|p-q|}
\sigma_p \sigma_q$ (with $\le 0$ what is actually measured).
\end{remark}

\begin{lemma}[Jensen-deficit identity]\label{lem:jensen} \PROVED{}
Let $J_p(c) = -\tfrac13\sum_t \log_2\!\big(3V_{p+1}(c_t)/\sum_s V_{p+1}(c_s)\big)
\ge 0$ be the within-block Jensen deficit (zero iff the three sub-blocks are
equal). Then, exactly,
\[
\E\big[\tilde X_{p+1} \,\big|\, \text{scale-}p \text{ block}\big] \;=\; -\,J_p,
\qquad\text{hence}\qquad
\mathrm{Cov}(\tilde X_p, \tilde X_{p+1}) \;=\; -\,\mathrm{Cov}(\tilde X_p, J_p).
\]
(Machine check: identity to $8\times 10^{-15}$; $J \ge 0$ everywhere;
$\max J = 0.119$ at $k{=}13$, an order of magnitude below the crude bound
$C_-^2/2$.) Consequently: (i) the lag-1 cross-term obeys
$|\mathrm{Cov}(\tilde X_p, \tilde X_{p+1})| \le \sigma_p\,\sigma(J_p)$ with
$\sigma(J_p) \le \max J_p$ bounded by the saturated spread
(Lemma~\ref{lem:pos}(ii)) --- an explicit finite $c_0$, which is all the
Summation Theorem requires; (ii) the measured \emph{sign} is explained:
$\mathrm{corr}(\tilde X_p, J_p) = +0.53$ --- rich blocks are internally more
spread --- so the cross-terms are negative, and proving this
positive-association (FKG-flavored, via the multiplicative cascade) would
upgrade the bound to the clean $\mathrm{Cov} \le 0$.
\end{lemma}

\begin{proposition}[Tower profile bound; Law B]\label{prop:towerB}
For every $1 \le p \le k-1$,
\[
\Var_{\mathrm{count}}(\tilde X_p) \;\le\; C_{\mathrm{inj}}\,
\mathrm{env}^{\,p-1},
\qquad
C_{\mathrm{inj}} = \Var_{\text{types}}\Big(\log_2 \tfrac1{1-\pbar_t}\Big),
\quad \mathrm{env} = \tfrac{B_1+B_3}{3\rho}.
\]
\emph{Proof outline (three proved ingredients + elasticity bookkeeping):}
(a) \emph{No injection at the top digit} (Lemma~\ref{lem:pos}(ii)): the
sibling triple at digit $p{-}1$ shares its coefficients exactly, so
$\tilde X_p$ is driven only through downstream differences. (b) By the
Block-Equation Lemma the difference field satisfies the \emph{same} linear
recursion as $V_p$ itself: backbone images are again digit-$(p{-}1)$
sibling triples, feed images are digit-$(p{-}2)$ sibling triples one level
down --- each feed descent lowers the differing digit by exactly one, so
the injection (differing branch coefficients) is reached only after
$p{-}1$ descents. (c) The counting mass of $g$-fold feed descents is the
proved envelope $\mathrm{env}^{g}$ (Corollary~\ref{cor:envelope}), and the
injected log-lift second moment is $C_{\mathrm{inj}}$ by definition.
(d) \emph{Antichain elasticity, explicit:} in the frozen linear system
$V_p(c)$ is \emph{linear} in the injected endpoint values jointly, so
$V_p(c) = \sum_j \tilde w_j I_j$ with positive path weights, and the
elasticities $e_j = \tilde w_j I_j / V_p(c)$ are positive and sum to
exactly $1$. By (a)--(b), siblings share every coefficient on paths
with fewer than $p{-}1$ feed descents: $V_p(c_t) = S + D_t$ with $S$
shared, so
$\tilde X_p = \log_2\big(1 + (D_t - \bar D)/(S + \bar D)\big)$ is
controlled by the \emph{deep elasticity}
$e_{\ge p-1}(c) = \bar D/V \in [0,1]$ times the relative spread of the
deep injections (whose log-variance is $C_{\mathrm{inj}}$; the Jensen
gap between log and linear spread is bounded by
Lemma~\ref{lem:pos}). Then
$\Var(\tilde X_p) \lesssim \E[e_{\ge p-1}^2]\,C_{\mathrm{inj}}
\le \E[e_{\ge p-1}]\,C_{\mathrm{inj}}$ using $e \le 1$. The one
residual link is the \emph{envelope-to-elasticity comparison}
$\E_{\mathrm{count}}[e_{\ge p-1}] \le \mathrm{env}^{p-1}$: per descent
the elasticity is the class feed share $\varphi$, whose counting mean
is $0.48 \ll \mathrm{env}$, and the product bookkeeping is
Corollary~\ref{cor:envelope}'s proved statement for the $b$-weights ---
converting $b$-products to $\varphi$-products is the last sentence.
\MEASURED{} (script 194\_tower\_task2.py, $k = 11,\dots,15$, every level):
the bound holds at \emph{every} $(k,p)$ with margin $\ge 1.76$ (attained
at $p=1$), growing to $7$--$15\times$ deep in the tower; the measured
per-level ratio plateaus at $0.65$--$0.73 \approx \pbar = 1 - \lam^{-2}$,
strictly below $\mathrm{env} = 0.731$--$0.735$ --- the observed decay rate
is the flow feed-share, the proved envelope sits above it. The $p=1$
margin decreases slowly in $k$ ($1.88 \to 1.76$ over $k=11..15$) with
shrinking increments (saturation shape), consistent with a limit well
above $1$; this creep is the same fine-end saturation already quantified
by $CV_1(k) = 0.5136 - 0.337\cdot 0.910^k$ (companion log, Prop.\ 23).
\end{proposition}

\begin{remark}[What this program says about the exceptional set --- and what
it cannot say]\label{rem:S}
Let $S$ be the set of integers not (yet) proven to reach $1$. The present
program makes the \emph{proven-convergent} set as thick as the method
allows ($x^{1-\varepsilon}$); it does not bound $|S|$ from above --- the
strongest upper bound remains logarithmic density zero (Tao 2019 plus
verification to $2^{71}$). Moreover the K--L machinery is
\emph{attractor-agnostic}: Theorem 2.2 bounds $\pi_a(x)$ for every
$a \not\equiv 0 \bmod 3$, so if a second cycle existed, \emph{its} basin
would enjoy the same $x^{\gamma}$ lower bounds. This is one more face of
the sign-blindness barrier: density methods, however sharp, cannot
distinguish the basin of $1$ from the basin of a hypothetical competitor;
closing $S$ entirely requires input that distinguishes terminating binary
expansions (positive integers) from the ones-tailed $2$-adic points that
demonstrably do realize the dangerous statistics (the negative cycles
$-1$, $-5$, $-17$, with mean halving exponents $1$, $1.5$, $1.571$, all
below $\log_2 3$).
\end{remark}

\begin{lemma}[Density beats depth]\label{lem:depth} \PROVED{}
One suppression level costs one fresh trit of counting density
(Lemma~\ref{lem:fresh} / Lemma~\ref{lem:sat}) and buys at most
$\log_2(\rho\lam^2) \le 1.24$ bits of downward deviation (the one-level
suppression factor is at least $\lam^{-2}/\rho$, directly from the equation).
Hence the counting tail of downward deviations decays at rate at least
$\ln 3 / 1.24 = 0.886$ nats per bit.
\end{lemma}

\begin{proposition}[Counting variance]\label{prop:vcount}
Combining the tower pieces: the summation bound
$\Var_{\mathrm{count}}(F) \le \sum_p \Var(\tilde X_p)$ (cross-terms
one-sidedly bounded via the Jensen-deficit identity,
Lemma~\ref{lem:jensen}, and measured $\le 0$ without exception) together
with the profile bound (Proposition~\ref{prop:towerB}) gives the fully
explicit
\[
\Var_{\mathrm{count}}(F) \;\le\; \sum_{p\ge 1} C_{\mathrm{inj}}\,
\mathrm{env}^{\,p-1} \;=\; \frac{C_{\mathrm{inj}}}{1-\mathrm{env}}
\;=\; 5.4 \text{ bits}^2 \quad (k=13),
\]
uniformly in $k$ (the constant drifts only through $\lam^*(k) \le 2$:
at the endpoint $C_{\mathrm{inj}}/(1-\mathrm{env}) \to 6.1$).
Measured: $\sum_p \Var(\tilde X_p) = 1.58$ and
$\Var_{\mathrm{count}}(F) = 1.38$ bits$^2$ at $k=13$ --- the bound holds
with $3.4\times$ headroom on the sum and $3.9\times$ on the variance
itself, and no longer requires the separate correlation-decay constant
$K_{\mathrm{orth}}$.
\end{proposition}

\emph{strictly decreasing} in $g$ everywhere --- the shells decay
faster than the envelope. The fine structure corrects a naive guess:
the per-shell tilt $W_g/\mathrm{count}_g$ is \emph{not} flat but grows
$\approx \times 1.3$ per level (flow concentrates on chains), and it is
paid for by the counting mass decaying at $\approx 0.30$ per level ---
much thinner than the structural $(2/3)^g$, because domination is a
strictly stronger event than chain existence. Per level:
counting surplus $2^{-1.30}$ versus tilt growth $2^{+0.38}$, the
shell-level face of density-beats-tilt (Lemma~\ref{lem:depth}).
\emph{What the proved constants give and what remains:} domination
chains are $\sigma$-chains, so the per-level \emph{counting} ratio is
$\le 2/3$ outright (Lemma~\ref{lem:fresh}); the measured counting ratio
$\approx 0.30$ shows the extra factor $\approx 0.45$ is the cost of
\emph{maintaining} the deviation across the feed step --- note the
factors do \emph{not} multiply as independent events (deviations are
persistent: a naive fresh-deviation cost per level would give $0.067$,
below the measured $0.30$ --- overclaiming refuted before it was made).
The chain therefore closes iff the \emph{tilted maintenance factor}
(flow-weighted one-step persistence of $\{G \le -t_0\}$ along the
selected feed edge) is $< \mathrm{env}/(2/3) = 1.10$. Measured:
$0.60$--$0.78$ ($= r(g)/(2/3)$), margin $\ge 1.4\times$, uniform in
$(k, \varepsilon, g)$. Bounding it is a per-level application of the
Task-3 tilt machinery to the one-step increment of $G$ --- whose
conditional variance the measurements already show \emph{contracts}
($\le 0.90$). This single inequality is the remaining content of
Task~4.
The Markov step is directly supported: conditioning on domination at the
current chain level \emph{contracts} the next level's variance (measured
conditional variance ratios $\le 0.90$, all $(k,\varepsilon,g)$) and
shifts its mean by $\le 0.94$ bits. \MEASURED{} realized per-step chain
ratios $r(g) = 0.40$--$0.52$, decaying in $g$, at every $k \le 15$ ---
uniformly below $\mathrm{env} = 0.731$--$0.735$ with $\ge 1.4\times$
per-step margin (honest note: $r(1)$ creeps up with $k$, $0.46 \to 0.52$
at $\varepsilon = 0.10$, the familiar fine-end creep; the envelope margin
persists).
\end{proposition}

\begin{lemma}[Transfer constants]\label{lem:transfer} \PROVED{}
(script 196\_transfer\_constants.py)
\emph{(i) Samuelson leg.} For every $k$,
\[
1 - q_k \;\le\; \sqrt2\;\mathrm{CV}_{w,\mathrm{top}}(k),
\]
where $\mathrm{CV}_{w,\mathrm{top}}$ is the flow-weighted quadratic mean
of the top-scale triple CVs. \emph{Proof (three lines):} Samuelson's
inequality for $n=3$ gives $\mathrm{mean} - \min \le \sqrt2\,\sigma$ for
every refinement triple; summing with weight $=$ triple mean gives
$1 - q_k = \sum(\mathrm{mean}-\min)/\sum\mathrm{mean} \le
\sqrt2 \sum\sigma/\sum\mathrm{mean}$; Cauchy--Schwarz converts the
weighted mean of CVs into $\mathrm{CV}_w$. (Verified $k=11..15$: zero
per-triple violations; aggregate margin $\approx 1.3\times$; the
empirical linearization constant $c_1 \in [1.07, 1.45]$ of the companion
log was Samuelson all along.)
\emph{(ii) Edge-rate leg, one-sided.} The Min-Loss identity forces every
critical pair onto the \emph{explicit} curve
$q(\lam) = 3(1-\lam^{-2})/(\lam^{\alpha-2}+\lam^{\alpha-1})$ (machine
check: certificate $q_k$ agrees to $10^{-12}$ at $k = 11..15$), so the
transfer is one-variable calculus:
\[
1 - \gamma \;\le\; \frac{1-q(\lam)}{\ln(4/3)}
\qquad\text{on all of } \lam \in (1,2],
\]
with equality only at the endpoint $\lam = 2$ (grid check: $2\times 10^6$
points, $\min h = 0$ attained at $\lam = 2$; the endpoint derivative
$d\gamma/dq = 1/\ln(4/3)$ is proved in \cite{G1}). Combining (i)+(ii):
\[
1 - \gamma_k \;\le\; \frac{\sqrt2}{\ln(4/3)}\,
\mathrm{CV}_{w,\mathrm{top}}(k) \;=\; 4.917\,\mathrm{CV}_{w,\mathrm{top}}(k),
\]
so $\mathrm{CV}_{\mathrm{top}}(k) \to 0$ transfers to $\gamma_k \to 1$
with fully explicit constants.
\end{lemma}

\begin{corollary}[Main chain]\label{cor:chain}
Lemma D $\Rightarrow$ flow-$L^2$ attenuation $\kappa \le \kappa_{\max} < 1$
$\Rightarrow$ $\mathrm{CV}_{\mathrm{top}}(k) \to 0$ geometrically
$\Rightarrow$ $q(k) \to 1$ $\Rightarrow$ $\gamma(k) \to 1$, the last two
implications now carrying proved constants (Lemma~\ref{lem:transfer}).
\end{corollary}

\section{The endpoint route}

Assembling the transfer constants exposed a shorter conditional route,
which we record because it changes what the open core \emph{is}.

\begin{theorem}[Endpoint route]\label{thm:endpoint}
Suppose \emph{Endpoint Decay} holds: there are $C$ and $r < 1$ with
\[
\Var_{\mathrm{count}}\big(\tilde X_{k-1}\big) \;\le\; C\,r^{\,k}
\qquad\text{for all } k,
\]
i.e.\ the \emph{single scalar sequence} of top-scale tower variances
decays geometrically in depth. Then $\gamma_k \to 1$, with explicit
constants:
\[
1 - \gamma_k \;\le\; \frac{\sqrt2}{\ln(4/3)}\,
\sqrt{C^{\mathrm{e}}_{\mathrm{tilt}}\, C}\;r^{k/2}\,(1 + O(\text{spread})).
\]
\emph{Proof.} The endpoint tower variance controls the linear top-triple
CVs up to the bounded Jensen gap (Lemma~\ref{lem:pos}); passing from
counting to flow weighting costs the endpoint tilt factor
$C^{\mathrm{e}}_{\mathrm{tilt}}$ (Proposition~\ref{prop:tilt}; measured
$\le 1.5$, and the flow-weighted CV is in fact \emph{smaller} than the
counting CV at $k \le 15$); Samuelson (Lemma~\ref{lem:transfer}(i))
converts $\mathrm{CV}_{w,\mathrm{top}}$ into $1 - q_k$; the explicit
min-loss curve (Lemma~\ref{lem:transfer}(ii)) converts $1-q_k$ into
$1-\gamma_k$. \qed

\emph{No Summation Theorem, no chain recursion, and no multi-scale
uniformity enter this implication.} The critical path of the whole
program reduces to one scalar sequence.
\end{theorem}

\[
\textsc{Endpoint Decay} \iff \limsup_k d_k < 1,
\]

\emph{Two-constants audit:} Proposition~\ref{prop:towerB}'s bound
